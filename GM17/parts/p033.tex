\stmt
(11.5) \textit{For any separation $(A, B)$ of $G$,}
\begin{equation*}
|D(A, B)| \le |V(A \cap B) - (S \cup \gamma(N(\Gamma)))|.
\end{equation*}
\textit{Proof.} For each $d \in D(A, B)$, there exists
\begin{equation*}
v_d \in V(A \cap B) \cap (V(\alpha(d)) - (S \cup V(\alpha(-d))))
\end{equation*}
by (11.4). Since $v_d \in V(\alpha(d)) - V(\alpha(-d))$, it follows that the vertices $v_d$ ($d \in D(A, B)$) are all distinct. Moreover, each $v_d$ belongs to $V(A \cap B)$, and not to $S \cup \gamma(N(\Gamma))$, since $\gamma(N(\Gamma)) \subseteq V(\alpha(-d))$. Hence
\begin{equation*}
|D(A, B)| = |\{v_d \colon d \in D(A, B)\}| \le |V(A \cap B) - (S \cup \gamma(N(\Gamma)))|,
\end{equation*}
as required. $\square$
