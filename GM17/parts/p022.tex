\sectionhead{9. \textsc{Cell} \textsc{Connectivity} \textsc{In} A \textsc{True} \textsc{Portrayal}}

Again, throughout this section $\pi = (\Sigma, \Gamma, \alpha, \beta, \gamma)$ is a true, $(2p + 3)$-redundant, $\mathcal{T}$-central portrayal of a hypergraph $G$, with warp $\le p$, and $ord(\mathcal{T}) > 4p + 5$. Now we study connectivity properties of the $\alpha(c)$'s for individual cells $c$.

\stmt
(9.1) \textit{Let $c_0$ be an internal cell, and let $v_1, v_2 \in \gamma(\tilde{c}_0)$. Then there is a path in $\alpha(c_0)^*$ from $v_1$ to $v_2$ with no other vertex in $\gamma(\tilde{c}_0)$.}

\textit{Proof.} Otherwise there is a separation $(A_1, A_2)$ of $\alpha(c)$ with $v_i \in V(A_i)$ $(i = 1, 2)$ and $V(A_1 \cap A_2) \subseteq \gamma(\tilde{c}_0) - \{v_1, v_2\}$. Hence $|V(A_1 \cap A_2)| \le 1$. Choose discs $\Delta_1, \Delta_2 \subseteq \bar{c}_0$ such that
\begin{align*}
\Delta_1 \cap \Delta_2 &= \{n \in \tilde{c}_0 : \gamma(n) \in V(A_1 \cap A_2)\} \\
\Delta_i &\supseteq \{n \in \tilde{c}_0 : \gamma(n) \in V(A_i)\} \quad (i = 1, 2).
\end{align*}
Let $c_i = \Delta_i - N(\Gamma)$ $(i = 1, 2)$, and let $\Gamma' = ((U(\Gamma) - c_0) \cup \Delta_1 \cup \Delta_2, N(\Gamma))$. Then $\Gamma'$ is a painting in $\Sigma$ with $C(\Gamma') = (C(\Gamma) - \{c_0\}) \cup \{c_1, c_2\}$. Define
\begin{align*}
\alpha'(c_1) &= A_1, \quad \alpha'(c_2) = A_2 \\
\alpha'(c) &= \alpha(c) \quad (c \in C(\Gamma') - \{c_1, c_2\}).
\end{align*}
Then $(\Sigma, \Gamma', \alpha', \beta, \gamma)$ is a $\mathcal{T}$-central portrayal of $G$ resembling $\pi$ but truer, a contradiction. $\square$
