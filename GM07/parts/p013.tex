\stmt
(4.6) \textit{Let $\Sigma$ be a connected surface with at least two cuffs and let $k \ge 0$ be an integer. Let $G$ be a proper graph in $\Sigma$, with $|V(G) \cap bd(\Sigma)| \le 2k$, such that}

\textit{(i) for every $G$-normal $I$-arc $B$ with $|V(G) \cap B| \le v(\Sigma, k)$ the ends of $B$ are in the same cuff of $\Sigma$,}

\textit{(ii) for every $G$-normal $O$-arc $B$ with $|V(G) \cap B| \le v(\Sigma, k)$ either $B$ is null-homotopic, or for some cuff $C$, $B$ is null-homotopic in $\Sigma + \hat{C}$ and $V(G) \cap C \subseteq B$, and}

\textit{(iii) for every cuff $C$, $|V(G) \cap C| \ge 2$.}

\textit{Let $H$ be a forest in $\Sigma$ with $V(G) \cap bd(\Sigma) = V(H) \cap bd(\Sigma)$. Then $H$ is $G$-feasible.}
