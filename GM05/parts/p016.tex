\stmt
(7.2) \textit{If $G\in \mathcal{F}_\theta$ and $X\subseteq V(G)$ then $G$ has a separation $(V_1, V_2)$ such that $|V_1\cap V_2|\le \theta_7$ and $|(V_1 - V_2)\cap X|, |(V_2 - V_1)\cap X|\le (1 - \theta_8^{-1})|X|$.}

\textit{Proof.} Let $N$ be an integer so large that there is no integer greater than $(1 - \theta_8^{-1})|X|$ and not greater than
\begin{equation*}
(1 - \theta_8^{-1})|X| + N^{-1}((1 - \theta_8^{-1})|V(G)| + \theta_7).
\end{equation*}
For each $v\in X$, let $N_v$ be a set of $N$ new vertices, and let $G'$ be the graph with vertex set
\begin{equation*}
V(G)\cup \bigcup_{v\in X} N_v
\end{equation*}
such that $G'\setminus \bigcup_{v\in X} N_v = G$, and for each $v\in X$, every $u\in N_v$ is adjacent in $G'$ to $v$ and to no other vertex. Evidently $G'\in \mathcal{F}_\theta$, and so by (7.1) there is a separation $(V_1', V_2')$ of $G'$ with $|V_1'\cap V_2'|\le \theta_7$ and
\begin{equation*}
|V_1' - V_2'|, |V_2' - V_1'| \le (1 - \theta_8^{-1})|V(G')|.
\end{equation*}
